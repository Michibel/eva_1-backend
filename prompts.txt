================================================================================
                    COMANDOS IMPORTANTES
================================================================================

# Configuración inicial
python -m venv env
.\env\Scripts\activate
pip install Django djangorestframework

# Crear proyecto y app
django-admin startproject academic_project .
python manage.py startapp academic

# Migraciones
python manage.py makemigrations
python manage.py migrate

# Crear super usuario
python manage.py createsuperuser

# Cargar datos
python manage.py load_initial_data

# Ejecutar servidor
python manage.py runserver

# URLs de acceso
http://127.0.0.1:8000/        - Página de inicio
http://127.0.0.1:8000/courses/ - Lista de cursos
http://127.0.0.1:8000/students/ - Lista de estudiantes
http://127.0.0.1:8000/admin/  - Panel administrativo
http://127.0.0.1:8000/api/    - API Root

================================================================================
                    ESTRUCTURA DE LA API
================================================================================

GET    /api/teachers/         - Lista de profesores
POST   /api/teachers/         - Crear profesor
GET    /api/teachers/{id}/    - Detalle de profesor
PUT    /api/teachers/{id}/    - Actualizar profesor
DELETE /api/teachers/{id}/    - Eliminar profesor

GET    /api/students/         - Lista de estudiantes
POST   /api/students/         - Crear estudiante
GET    /api/students/{id}/    - Detalle de estudiante
PUT    /api/students/{id}/    - Actualizar estudiante
DELETE /api/students/{id}/    - Eliminar estudiante

GET    /api/courses/          - Lista de cursos
POST   /api/courses/          - Crear curso
GET    /api/courses/{id}/     - Detalle de curso
PUT    /api/courses/{id}/     - Actualizar curso
DELETE /api/courses/{id}/     - Eliminar curso

GET    /api/enrollments/      - Lista de inscripciones
POST   /api/enrollments/      - Crear inscripción
GET    /api/enrollments/{id}/ - Detalle de inscripción
PUT    /api/enrollments/{id}/ - Actualizar inscripción
DELETE /api/enrollments/{id}/ - Eliminar inscripción
